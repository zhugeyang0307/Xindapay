二、首次部署

2.1 本地打包

cd D:\XindaPay_mail\pay.yunzexinda.com
tar -czf D:\deploy.tar.gz `
  --exclude='*.bak*' --exclude='_test*' --exclude='_dev*' `
  --exclude='_sim*' --exclude='_ft*' --exclude='_gen*' `
  --exclude='_reset*' --exclude='_pb*' --exclude='_fulltest*' `
  --exclude='node_modules' --exclude='runtime/*' --exclude='*.log' `
  --exclude='config.php' --exclude='telegram/config.php' .

2.2 上传

scp -P 47395 -i \C:\Users\tytdw\.ssh\id_ed25519 `
  D:\deploy.tar.gz jinmei:/tmp/deploy.tar.gz

2.3 线上解压

ssh server124
cd /www/wwwroot/pay.yunzexinda.com
tar -czf /www/backup/pay_\.tar.gz .
tar -xzf /tmp/deploy.tar.gz --exclude='config.php' --exclude='telegram/config.php'
chown -R www:www .
chmod -R 777 runtime/ uploads/

2.4 数据库更新

cd /www/server/mysql/bin
./mysql -uxindapay -pREPLACE_WITH_DB_PASSWORD xindapay < /tmp/migration.sql
migration.sql 内容 
ALTER TABLE pay_user
  ADD COLUMN service_expire DATETIME NULL,
  ADD COLUMN service_trial_end DATETIME NULL,
  ADD COLUMN apk_quota INT UNSIGNED NOT NULL DEFAULT 3,
  ADD COLUMN sub_level TINYINT NOT NULL DEFAULT 1,
  ADD COLUMN sub_push_tg VARCHAR(32),
  ADD COLUMN sub_push_email VARCHAR(64),
  ADD COLUMN sub_push_smtp_host VARCHAR(100),
  ADD COLUMN sub_push_smtp_user VARCHAR(100),
  ADD COLUMN sub_push_smtp_pwd VARCHAR(255),
  ADD COLUMN sub_push_smtp_secure VARCHAR(10) DEFAULT 'ssl',
  ADD COLUMN sub_push_status TINYINT DEFAULT 0;

ALTER TABLE pay_alipay_bill
  ADD COLUMN is_deleted TINYINT NOT NULL DEFAULT 0,
  ADD COLUMN deleted_by INT UNSIGNED DEFAULT 0,
  ADD COLUMN deleted_at DATETIME NULL;

INSERT INTO pay_config (k,v) VALUES
  ('sys_month_price','100'), ('sys_trial_days','2'),
  ('sys_grace_days','0'), ('sys_apk_quota','3'),
  ('sys_extra_device_price','50')
ON DUPLICATE KEY UPDATE v=VALUES(v);

UPDATE pay_user SET service_expire = DATE_ADD(NOW(), INTERVAL 10 YEAR)
  WHERE service_expire IS NULL;

2.5 重启 PHP

service php-fpm-82 restart
